message(STATUS "Configure CMakeCompileOpticalflowShaders.txt")
message(STATUS "FFX_GLSLANG_EXE_PATH=${FFX_GLSLANG_EXE_PATH}")

set(ARM_OPTICALFLOW_BASE_ARGS
    -glslangexe=${FFX_GLSLANG_EXE_PATH}
    -reflection -deps=gcc -DMIN_SAD -DFFX_GPU=1)

if(WIN32)
    set(ARM_OPTICALFLOW_PERMUTATION_ARGS
        -DDEPTH_INVERTED={0,1}
    )
else()
    # need to add quotes around the values to avoid the linux shell
    # expand the args automatically
    set(ARM_OPTICALFLOW_PERMUTATION_ARGS
        -DDEPTH_INVERTED="{0,1}"
    )
endif()

set(ARM_OPTICALFLOW_INCLUDE_ARGS
	"${FFX_GPU_PATH}"
	"${FFX_GPU_PATH}/opticalflow")

set(ARM_OPTICALFLOW_API_COMMON_ARGS
    -compiler=glslang -e main --target-env vulkan1.1 -Os -DFFX_GLSL=1)

file(GLOB ARM_OPTICALFLOW_SHADERS_COMP
    "shaders/opticalflow/ffx_opticalflow_mv_hints.glsl"
)

file(GLOB ARM_OPTICALFLOW_SHADERS_FRAG
    "shaders/opticalflow/ffx_opticalflow_mv_hints_fs.glsl"
)

file(GLOB ARM_OPTICALFLOW_SHADERS_VERT
    "shaders/opticalflow/ffx_opticalflow_mv_hints_vs.glsl"
)

compile_all_shaders(
    "${FFX_SC_EXECUTABLE}"
    "${ARM_OPTICALFLOW_BASE_ARGS}" "${ARM_OPTICALFLOW_API_COMMON_ARGS}"
    "${ARM_OPTICALFLOW_PERMUTATION_ARGS}" "${ARM_OPTICALFLOW_INCLUDE_ARGS}"
    "${ARM_OPTICALFLOW_SHADERS_COMP}" "${ARM_OPTICALFLOW_SHADERS_FRAG}" "${ARM_OPTICALFLOW_SHADERS_VERT}"
    "${FFX_PASS_SHADER_OUTPUT_PATH}" ARM_OPTICALFLOW_PERMUTATION_OUTPUTS
)

add_shader_output("${ARM_OPTICALFLOW_PERMUTATION_OUTPUTS}")
