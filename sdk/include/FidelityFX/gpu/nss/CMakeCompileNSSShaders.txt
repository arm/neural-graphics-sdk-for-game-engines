message(STATUS "Configure CMakeCompileNSSShaders.txt")
message(STATUS "FFX_GLSLANG_EXE_PATH=${FFX_GLSLANG_EXE_PATH}")

set(NSS_BASE_ARGS
    -glslangexe=${FFX_GLSLANG_EXE_PATH}
    -DOUTPUT_IMG_FORMAT=r11f_g11f_b10f
    -DFFX_8_BIT_TYPES=1
    -reflection -deps=gcc -DFFX_GPU=1
    -DQUANTIZED=1
    )

if(WIN32)
    set(NSS_PERMUTATION_ARGS
        -DREVERSE_Z={0,1}
        -DRESAMPLE_BICUBIC={0,1}
        -DSCALE_PRESET_MODE={0,1}
        -DMANAGE_HISTORY={0,1}
        -DNSS_SHADER_QUALITY_MODE={0,1,2}
        -DNSS_SUPPORT_TENSOR={0,1})

else()
    # need to add quotes around the values to avoid the linux shell
    # expand the args automatically
    set(NSS_PERMUTATION_ARGS
        -DREVERSE_Z="{0,1}"
        -DRESAMPLE_BICUBIC="{0,1}"
        -DSCALE_PRESET_MODE="{0,1}"
        -DMANAGE_HISTORY="{0,1}"
        -DNSS_SHADER_QUALITY_MODE="{0,1,2}"
        -DNSS_SUPPORT_TENSOR="{0,1}"
        )
endif()

set(NSS_INCLUDE_ARGS
    "${FFX_GPU_PATH}"
    "${FFX_GPU_PATH}/nss")

# =============================================================================
#      Build compute shader
# =============================================================================
set(NSS_API_BASE_ARGS_COMP
    -compiler=glslang -e main --target-env vulkan1.1 -S comp -DFFX_GLSL=1)

file(GLOB NSS_SHADERS_COMP
    "shaders/nss/*.glsl")
list(FILTER NSS_SHADERS_COMP EXCLUDE REGEX ".*_fs\\.glsl$")
list(FILTER NSS_SHADERS_COMP EXCLUDE REGEX ".*_vs\\.glsl$")

compile_shaders_with_depfile(
    "${FFX_SC_EXECUTABLE}"
    "${NSS_BASE_ARGS}" "${NSS_API_BASE_ARGS_COMP}" "${NSS_PERMUTATION_ARGS}" "${NSS_INCLUDE_ARGS}"
    "${NSS_SHADERS_COMP}" "${FFX_PASS_SHADER_OUTPUT_PATH}" NSS_PERMUTATION_OUTPUTS_COMP)

add_shader_output("${NSS_PERMUTATION_OUTPUTS_COMP}")

# =============================================================================
#      Build fragment shader
# =============================================================================
set(NSS_API_BASE_ARGS_FRAG
    -compiler=glslang -e main --target-env vulkan1.1 -S frag -DFFX_GLSL=1)

file(GLOB NSS_SHADERS_FRAG
    "shaders/nss/*_fs.glsl")

compile_shaders_with_depfile(
    "${FFX_SC_EXECUTABLE}"
    "${NSS_BASE_ARGS}" "${NSS_API_BASE_ARGS_FRAG}" "${NSS_PERMUTATION_ARGS}" "${NSS_INCLUDE_ARGS}"
    "${NSS_SHADERS_FRAG}" "${FFX_PASS_SHADER_OUTPUT_PATH}" NSS_PERMUTATION_OUTPUTS_FRAG)

add_shader_output("${NSS_PERMUTATION_OUTPUTS_FRAG}")

# =============================================================================
#      Build vertex shader
# =============================================================================
set(NSS_API_BASE_ARGS_VERT
    -compiler=glslang -e main --target-env vulkan1.1 -S vert -DFFX_GLSL=1)

file(GLOB NSS_SHADERS_VERT
    "shaders/nss/*_vs.glsl")

compile_shaders_with_depfile(
    "${FFX_SC_EXECUTABLE}"
    "${NSS_BASE_ARGS}" "${NSS_API_BASE_ARGS_VERT}" "${NSS_PERMUTATION_ARGS}" "${NSS_INCLUDE_ARGS}"
    "${NSS_SHADERS_VERT}" "${FFX_PASS_SHADER_OUTPUT_PATH}" NSS_PERMUTATION_OUTPUTS_VERT)

add_shader_output("${NSS_PERMUTATION_OUTPUTS_VERT}")

# data graph models

# for now, assume files will be in vgf format (but extension might be slightly different )
if (NOT NSS_DATA_GRAPH_EXT)
    set(NSS_DATA_GRAPH_EXT vgf)
endif()


file(GLOB NSS_DATA_GRAPHS
    "data_graphs/nss/*.${NSS_DATA_GRAPH_EXT}")

compile_data_graphs(
    "${VGF_PARSER_EXECUTABLE}"
    "${FFX_PASS_DATA_GRAPH_OUTPUT_PATH}/"
    "${NSS_DATA_GRAPHS}" 
    NSS_DATA_GRAPH_OUTPUTS)

# add the header files they generate to the main list of dependencies
add_shader_output("${NSS_DATA_GRAPH_OUTPUTS}")
